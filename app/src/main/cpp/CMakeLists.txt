cmake_minimum_required(VERSION 3.22.1)
project("nucleo_brain")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -O3 -ffast-math -fvisibility=hidden")
add_library(nucleo_brain SHARED brain.cpp jni_bridge.cpp)
target_include_directories(nucleo_brain PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
find_library(log-lib log)
target_link_libraries(nucleo_brain ${log-lib})
if(ANDROID_ABI STREQUAL "arm64-v8a")
    target_compile_options(nucleo_brain PRIVATE -march=armv8-a+fp+simd)
elseif(ANDROID_ABI STREQUAL "armeabi-v7a")
    target_compile_options(nucleo_brain PRIVATE -mfpu=neon)
endif()
